\begin{frame}
	\frametitle{Motivation}
	\framesubtitle{Designed vs. discovered RL rules}

	\begin{columns}[T]
		\begin{column}{0.48\linewidth}
			\textbf{The problem}

			Performant human-designed RL rules leverage knowledge about the dynamics of the system.

			Ex. REACT is designed (from the ground up) to exploit coherent flow structures in the wake; it makes use of knowledge about the varying timescales of turbulent phenomena.

			One posits that the meta-learned semantics should capture some inherent quality of this class of problem.
		\end{column}
		\begin{column}{0.48\linewidth}
			\textbf{Why it matters}
			\begin{itemize}
				\item \textbf{Generalisability:} discover rules on cheaper problems, learn the semantics, apply to harder problems.
				\item \textbf{Expressivity:} develop semantics that represent properties intangible to humans.
				\item \textbf{Performance:} empirically, discovered RL rules outperform their counterparts.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
